#  ★ 这里同时修掉「上一版固件无线崩溃」的两个根因 ★
#  Config/GENERAL.txt 是给 MT7986 / 6.6 分支写的，它有几行会污染 MT7987 的配置。
# ============================================================================

# --- 根因 1（致命）：GENERAL.txt L152 开了老的 C 版 iwinfo ---
#   iwinfo-ucode 的依赖写作 `+ucode-mod-iwinfo @!PACKAGE_iwinfo`（互斥）：
#     GENERAL: CONFIG_PACKAGE_iwinfo=y
#     -> iwinfo-ucode 被禁用
#     -> mtwifi-cfg-ucode 依赖不满足，被 kconfig 静默丢弃
#     -> luci-app-mtwifi-cfg 连锁丢弃
#     -> 只剩主线 mac80211.sh 去配置联发科闭源驱动（type 变成 mac80211、
#        band 被猜成 6g、country=00）
#     -> hostapd 起不来 + 一查无线信息就内核 panic 重启
#   实测证据：上一版固件 manifest 里有 iwinfo，但没有 iwinfo-ucode /
#   mtwifi-cfg-ucode / luci-app-mtwifi-cfg。
# CONFIG_PACKAGE_iwinfo is not set
CONFIG_PACKAGE_iwinfo-ucode=y
CONFIG_PACKAGE_ucode-mod-iwinfo=y
CONFIG_PACKAGE_mtwifi-cfg-ucode=y
CONFIG_PACKAGE_luci-app-mtwifi-cfg=y

# --- 根因 2：GENERAL.txt L370 把 WARP 版本覆盖成了 MT7986 的 2 ---
#   warp/config.in 明确：MT7987 的默认是 WARP_CONFIG_MODE=1、WARP_VERSION="3_1"
#   （帮助里写着 >=WARP_V3_1 必须 mode=1）。被覆盖成 2 会构建出错版的
#   mtk_warp.ko，而它管的正是无线的内存/环缓冲
#   -> 野指针 -> mt_cfg80211_get_wiphy 里 panic。
CONFIG_WARP_CONFIG_MODE=1
CONFIG_WARP_VERSION="3_1"
CONFIG_WARP_CHIPSET="mt7987"
CONFIG_WARP_DBG_SUPPORT=y
# CONFIG_WARP_MEMORY_LEAK_DBG is not set
# WED_HW_RRO_SUPPORT 依赖 WARP_CONFIG_MODE=0，MT7987 是 1，本就会被忽略
# CONFIG_WED_HW_RRO_SUPPORT is not set

# --- 风扇控制插件（作者那份 defconfig 里没有，是他发固件时另加的）---
#   依赖：+luci-base +rpcd-mod-ucode +kmod-hwmon-pwmfan
CONFIG_PACKAGE_luci-app-fancontrol-mtk=y
CONFIG_PACKAGE_kmod-hwmon-pwmfan=y

# --- 核心目的：ksmbd（内核态 SMB 服务）---
CONFIG_PACKAGE_kmod-fs-ksmbd=y
CONFIG_PACKAGE_ksmbd-server=y
CONFIG_PACKAGE_luci-app-ksmbd=y
CONFIG_PACKAGE_luci-i18n-ksmbd-zh-cn=y

# --- 关掉 Samba4（smbd 和 ksmbd 抢 445 端口）---
# CONFIG_PACKAGE_luci-app-samba4 is not set
# CONFIG_PACKAGE_samba4-server is not set

# --- 排障工具 ---
CONFIG_PACKAGE_lsof=y
