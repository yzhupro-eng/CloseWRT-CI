# --- 核心目的：ksmbd（内核态 SMB 服务）---
# 原来的问题：Samba4 是用户态服务，数据要在内核态/用户态之间来回拷贝，
# 4×A53@2.0GHz 上基本就卡在 ~50MB/s；ksmbd 在内核里直接服务 SMB，
# 同硬件通常能到 150~250MB/s（前提是磁盘和链路不是瓶颈）。
CONFIG_PACKAGE_kmod-fs-ksmbd=y
CONFIG_PACKAGE_ksmbd-server=y
CONFIG_PACKAGE_luci-app-ksmbd=y
CONFIG_PACKAGE_luci-i18n-ksmbd-zh-cn=y

# --- 关掉 Samba4 ---
# Config/GENERAL.txt 里默认开了 luci-app-samba4；samba4 的 smbd 和 ksmbd
# 都要占 445 端口，两个一起装会互相打架，所以这里显式关掉，只留 ksmbd。
# CONFIG_PACKAGE_luci-app-samba4 is not set
# CONFIG_PACKAGE_samba4-server is not set

# --- 排障要用的工具（看句柄占用、看盘健康）---
CONFIG_PACKAGE_lsof=y
